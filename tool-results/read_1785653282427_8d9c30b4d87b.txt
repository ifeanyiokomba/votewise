     1→// VoteWise SUG v2 — Client-side API helpers.
     2→// Officials authenticate via HttpOnly cookies (set by /api/auth/login) → no
     3→// token management needed; `credentials: 'include'` carries the cookies.
     4→// Voters still use a header token (x-voter-token) for their session.
     5→
     6→const VOTER_TOKEN_KEY = 'votewise_voter_token'
     7→
     8→export function getVoterToken() { return typeof window === 'undefined' ? null : localStorage.getItem(VOTER_TOKEN_KEY) }
     9→export function setVoterToken(t: string | null) {
    10→  if (typeof window === 'undefined') return
    11→  if (t) localStorage.setItem(VOTER_TOKEN_KEY, t); else localStorage.removeItem(VOTER_TOKEN_KEY)
    12→}
    13→
    14→async function req<T = any>(path: string, opts: RequestInit = {}, voterToken?: string | null): Promise<T> {
    15→  const headers: Record<string, string> = { 'content-type': 'application/json', ...(opts.headers as Record<string, string> || {}) }
    16→  if (voterToken) headers['x-voter-token'] = voterToken
    17→  // Officials: cookies are sent automatically with credentials:'include'.
    18→  const res = await fetch(path, { ...opts, headers, credentials: 'include' })
    19→  const text = await res.text()
    20→  let data: any = null
    21→  try { data = text ? JSON.parse(text) : null } catch { data = { raw: text } }
    22→  if (!res.ok) {
    23→    const msg = data?.error || `Request failed (${res.status})`
    24→    // Provide user-friendly messages for auth/permission errors
    25→    const friendlyMsg = res.status === 401
    26→      ? 'Please sign in to continue.'
    27→      : res.status === 403
    28→      ? 'You do not have permission to perform this action. Please sign in as an admin.'
    29→      : msg
    30→    const err: any = new Error(friendlyMsg)
    31→    err.status = res.status
    32→    err.data = data
    33→    err.originalMessage = msg
    34→    throw err
    35→  }
    36→  return data as T
    37→}

   427→  // Per-election export & reports — returns a direct-download URL for use
   428→  // with window.open() or <a href download>. The browser handles auth via
   429→  // the existing HttpOnly cookies (credentials:'include' isn't honoured by
   430→  // window.open, but the cookies are sent automatically for same-origin
   431→  // navigations, which is what we want here).
   432→  exportElectionData: (
   433→    electionId: string,
   434→    type: 'results' | 'audit' | 'voters' | 'full',
   435→    format: 'csv' | 'json' | 'printable',
   436→    subdomain?: string,
   437→  ) => {
   438→    const origin = typeof window !== 'undefined' ? window.location.origin : ''
   439→    const q = new URLSearchParams()
   440→    q.set('type', type)
   441→    q.set('format', format)
   442→    if (subdomain) q.set('x-vw-org', subdomain)
   443→    return `${origin}/api/workspace/elections/${electionId}/export?${q.toString()}`
   444→  },
