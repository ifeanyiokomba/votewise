   158→function formatDateTime(iso: string | null | undefined): string {
   159→  if (!iso) return '—'
   160→  try {
   161→    return new Date(iso).toLocaleString(undefined, {
   162→      year: 'numeric', month: 'short', day: '2-digit',
   163→      hour: '2-digit', minute: '2-digit',
   164→    })
   165→  } catch {
   166→    return iso
   167→  }
   168→}
   169→
   170→function timeAgo(iso: string | null | undefined): string {
   171→  if (!iso) return 'Never'
   172→  try {
   173→    const diff = Date.now() - new Date(iso).getTime()
   174→    const sec = Math.max(0, Math.floor(diff / 1000))
   175→    if (sec < 5) return 'just now'
   176→    if (sec < 60) return `${sec}s ago`
   177→    const min = Math.floor(sec / 60)
   178→    if (min < 60) return `${min}m ago`
   179→    const hr = Math.floor(min / 60)
   180→    if (hr < 24) return `${hr}h ago`
   181→    const day = Math.floor(hr / 24)
   182→    if (day < 30) return `${day}d ago`
   183→    return formatDateTime(iso)
   184→  } catch {
   185→    return iso
   186→  }
   187→}
   188→
   189→function timeAgoShort(iso: string | null | undefined): string {
   190→  if (!iso) return '—'
   191→  try {
   192→    const diff = Date.now() - new Date(iso).getTime()
   193→    const day = Math.max(0, Math.floor(diff / 86_400_000))
   194→    if (day <= 0) return 'today'
   195→    if (day === 1) return '1d ago'
   196→    return `${day}d ago`
   197→  } catch {
   198→    return iso
   199→  }
   200→}

   209→  }

   213→  if (pct >= 99.9) return 'bg-emerald-500'

   224→function downtimeMinutes(pct: number): number {
   225→  // (100 - uptime%) of a 1440-min day → minutes of downtime
   226→  const m = Math.round((100 - pct) * 14.4)
   227→  return Math.max(1, m)
   228→}

   315→export function PlatformStatusPage() {
   316→  const [status, setStatus] = useState<PlatformStatus | null>(null)
   317→  const [history, setHistory] = useState<UptimeHistory | null>(null)
   318→  const [loading, setLoading] = useState(true)
   319→  const [loadingBars, setLoadingBars] = useState(true)
   320→  const [error, setError] = useState<string | null>(null)
   321→  const [lastUpdated, setLastUpdated] = useState<string>('')
   322→  const [countdown, setCountdown] = useState(30)
   323→  const [refreshing, setRefreshing] = useState(false)
   324→
   325→  // ------------------------------------------------------------------
   326→  // Data loaders
   327→  // ------------------------------------------------------------------
   328→
   329→  const loadStatus = useCallback(async (silent = false) => {
   330→    if (!silent) {
   331→      setRefreshing(true)
   332→      setError(null)
   333→    }
   334→    try {
   335→      const data = await api.pihedStatus() as PlatformStatus
   336→      setStatus(data)
   337→      setLastUpdated(new Date().toISOString())
   338→      if (!silent) setError(null)
   339→    } catch (e: unknown) {
   340→      if (!silent) {
   341→        setError(e instanceof Error ? e.message : 'Failed to load platform status')
   342→      }
   343→    } finally {
   344→      if (!silent) setRefreshing(false)
   345→      setLoading(false)
   346→    }
   347→  }, [])
   348→
   349→  const loadHistory = useCallback(async () => {
   350→    try {
   351→      const data = await api.pihedUptime(90) as { history?: UptimeHistory } | UptimeHistory
   352→      // Backend returns { history: {...} } — be defensive.
   353→      const hist = (data && 'history' in data && data.history) ? data.history : (data as UptimeHistory)
   354→      setHistory(hist || {})
   355→    } catch {
   356→      setHistory({})
   357→    } finally {
   358→      setLoadingBars(false)
   359→    }
   360→  }, [])
   361→
   362→  const refreshAll = useCallback(async () => {
   363→    await Promise.all([loadStatus(false), loadHistory()])
   364→    setCountdown(30)
   365→  }, [loadStatus, loadHistory])
   366→
   367→  // Initial load — runs once on mount.
   368→  useEffect(() => {
   369→    refreshAll()
   370→  }, [refreshAll])
   371→
   372→  // 30s auto-refresh + 1s countdown
   373→  useEffect(() => {
   374→    const tick = setInterval(() => {
   375→      setCountdown((c) => {
   376→        if (c <= 1) {
   377→          loadStatus(true) // silent refresh
   378→          return 30
   379→        }
   380→        return c - 1
   381→      })
   382→    }, 1000)
   383→    return () => clearInterval(tick)
   384→  }, [loadStatus])
   385→
   386→  // ------------------------------------------------------------------
   387→  // Derived data
   388→  // ------------------------------------------------------------------
   389→
   390→  const overall: OverallStatus = status?.status ?? 'OPERATIONAL'
   391→  const overallCfg = OVERALL_CONFIG[overall]
   392→  const services = status?.services ?? []
   393→  const incidents = status?.incidents ?? []
   394→  const maintenance = status?.maintenance ?? []
   395→  const overallUptime = status?.uptime ?? 99.99
   396→
   397→  // Build the incident-history timeline from the uptime bars: collect every
   398→  // day in the last 30 days with incidents > 0 across all tracked services.
   399→  const incidentDays = useMemo(() => {
   400→    if (!history) return []
   401→    const entries: Array<{ date: string; service: string; uptimePct: number; incidents: number }> = []
   402→    for (const [svc, series] of Object.entries(history)) {
   403→      // Take only the last 30 days
   404→      const last30 = series.slice(-30)
   405→      for (const day of last30) {
   406→        if (day.incidents > 0 || day.uptimePct < 100) {
   407→          entries.push({ date: day.date, service: svc, uptimePct: day.uptimePct, incidents: day.incidents })
   408→        }
   409→      }
   410→    }
   411→    // Sort: most recent first
   412→    entries.sort((a, b) => (a.date < b.date ? 1 : a.date > b.date ? -1 : 0))
   413→    return entries.slice(0, 10)
   414→  }, [history])
   415→
   416→  // ------------------------------------------------------------------
   417→  // Loading state — full-page skeleton
   418→  // ------------------------------------------------------------------
   419→
   420→  if (loading) {
   421→    return <StatusPageSkeleton />
   422→  }
   423→
   424→  // ------------------------------------------------------------------
   425→  // Error state — friendly retry card
   426→  // ------------------------------------------------------------------
   427→
   428→  if (error && !status) {
   429→    return (
   430→      <div className="mx-auto max-w-2xl px-4 py-16 sm:px-6">
   431→        <Card className="border-red-500/30 bg-red-500/5">
   432→          <CardContent className="flex flex-col items-center gap-4 p-8 text-center">
   433→            <div className="grid h-14 w-14 place-items-center rounded-full bg-red-100 dark:bg-red-500/15">
   434→              <ServerCrash className="h-7 w-7 text-red-600 dark:text-red-400" />
   435→            </div>
   436→            <div>
   437→              <h2 className="font-display text-xl font-semibold text-red-700 dark:text-red-300">
   438→                Status information unavailable
   439→              </h2>
   440→              <p className="mt-1 text-sm text-muted-foreground">
   441→                We couldn't reach the VoteWise health service. Please try again.
   442→              </p>
   443→            </div>
   444→            <Button onClick={() => refreshAll()} className="gap-2" disabled={refreshing}>
   445→              {refreshing ? <Loader2 className="h-4 w-4 animate-spin" /> : <RefreshCw className="h-4 w-4" />}
   446→              {refreshing ? 'Retrying…' : 'Retry'}
   447→            </Button>
   448→          </CardContent>
   449→        </Card>
   450→      </div>
   451→    )
   452→  }
   453→
   454→  return (
   455→    <div className="mx-auto max-w-6xl px-4 py-8 sm:px-6 lg:py-12">
   456→      {/* ============================ 1. HERO HEADER ============================ */}
   457→      <motion.header
   458→        initial={{ opacity: 0, y: -8 }}
   459→        animate={{ opacity: 1, y: 0 }}
   460→        transition={{ duration: 0.4 }}
   461→        className="mb-8"
   462→      >
   463→        <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   464→          <div className="flex items-center gap-3">
   465→            <div className="grid h-12 w-12 place-items-center rounded-xl bg-primary/10 text-primary">
   466→              <Shield className="h-6 w-6" />
   467→            </div>
   468→            <div>
   469→              <h1 className="font-display text-2xl font-bold tracking-tight sm:text-3xl lg:text-4xl">
   470→                VoteWise Platform Status
   471→              </h1>
   472→              <p className="mt-0.5 text-sm text-muted-foreground">
   473→                Real-time health and availability of all VoteWise services across{' '}
   474→                <span className="font-medium text-foreground">{REGION_COUNT} regions</span>.
   475→              </p>
   476→            </div>
   477→          </div>
   478→
   479→          <div className="flex items-center gap-2">
   480→            {/* LIVE pulse indicator */}
   481→            <div
   482→              className="flex items-center gap-2 rounded-full border border-emerald-500/30 bg-emerald-500/10 px-3 py-1.5"
   483→              aria-label="Live status feed"
   484→            >
   485→              <span className="relative flex h-2.5 w-2.5">
   486→                <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-emerald-400 opacity-75" />
   487→                <span className="relative inline-flex h-2.5 w-2.5 rounded-full bg-emerald-500" />
   488→              </span>
   489→              <span className="text-xs font-semibold uppercase tracking-wide text-emerald-700 dark:text-emerald-300">
   490→                Live
   491→              </span>
   492→            </div>
   493→
   494→            {/* Refresh button */}
   495→            <Button
   496→              variant="outline"
   497→              size="sm"

  1050→function SubscribeForm() {
  1051→  const [email, setEmail] = useState('')
  1052→  const [busy, setBusy] = useState(false)
  1053→
  1054→  function handleSubmit(e: React.FormEvent) {
  1055→    e.preventDefault()
  1056→    const trimmed = email.trim()
  1057→    if (!trimmed) {
  1058→      toast.error('Please enter your email address')
  1059→      return
  1060→    }
  1061→    // RFC-5322-lite email check
  1062→    const ok = /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(trimmed)
  1063→    if (!ok) {
  1064→      toast.error('Please enter a valid email address')
  1065→      return
  1066→    }
  1067→    setBusy(true)
  1068→    // Simulate a network call — UX touch only, no backend required.
  1069→    setTimeout(() => {
  1070→      setBusy(false)
  1071→      setEmail('')
  1072→      toast.success("Subscribed — you'll receive email alerts for incidents", {
  1073→        description: `We'll send platform health updates to ${trimmed}`,
  1074→      })
  1075→    }, 600)
  1076→  }
  1077→
  1078→  return (
  1079→    <form onSubmit={handleSubmit} className="flex w-full max-w-md items-center gap-2 sm:w-auto">
  1080→      <Input
  1081→        type="email"
  1082→        inputMode="email"
  1083→        autoComplete="email"
  1084→        placeholder="you@university.edu.ng"
  1085→        value={email}
  1086→        onChange={(e) => setEmail(e.target.value)}
  1087→        aria-label="Email address"
  1088→        className="flex-1 sm:w-64"
  1089→        disabled={busy}
  1090→      />
  1091→      <Button type="submit" disabled={busy} className="gap-1.5">
  1092→        {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <BellRing className="h-4 w-4" />}
  1093→        Subscribe
  1094→      </Button>
  1095→    </form>
  1096→  )
  1097→}
