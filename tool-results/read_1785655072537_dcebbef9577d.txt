   307→function formatDateTime(iso: string | null | undefined): string {
   308→  if (!iso) return '—'
   309→  try {
   310→    return new Date(iso).toLocaleString(undefined, {
   311→      year: 'numeric', month: 'short', day: '2-digit',
   312→      hour: '2-digit', minute: '2-digit',
   313→    })
   314→  } catch {
   315→    return iso
   316→  }
   317→}
   318→
   319→function timeAgo(iso: string | null | undefined): string {
   320→  if (!iso) return 'Never'
   321→  try {
   322→    const diff = Date.now() - new Date(iso).getTime()
   323→    const sec = Math.floor(diff / 1000)
   324→    if (sec < 60) return `${sec}s ago`
   325→    const min = Math.floor(sec / 60)
   326→    if (min < 60) return `${min}m ago`
   327→    const hr = Math.floor(min / 60)
   328→    if (hr < 24) return `${hr}h ago`
   329→    const day = Math.floor(hr / 24)
   330→    if (day < 30) return `${day}d ago`
   331→    return formatDateTime(iso)
   332→  } catch {
   333→    return iso
   334→  }
   335→}
   336→
   337→function formatUptime(sec: number): string {
   338→  if (!Number.isFinite(sec) || sec <= 0) return '—'
   339→  const d = Math.floor(sec / 86400)
   340→  const h = Math.floor((sec % 86400) / 3600)
   341→  const m = Math.floor((sec % 3600) / 60)
   342→  return `${d}d ${h}h ${m}m`
   343→}
   344→
   345→function formatNumber(n: number): string {
   346→  if (!Number.isFinite(n)) return '0'
   347→  return n.toLocaleString('en-NG')
   348→}
   349→
   350→function formatBytes(bytes: number): string {
   351→  if (!bytes) return '0 MB'
   352→  const mb = bytes / 1024 / 1024
   353→  if (mb < 1024) return `${mb.toFixed(1)} MB`
   354→  return `${(mb / 1024).toFixed(2)} GB`
   355→}
   356→
   357→function formatDuration(ms: number | null | undefined): string {
   358→  if (!ms) return '—'
   359→  if (ms < 1000) return `${ms}ms`
   360→  const s = ms / 1000
   361→  if (s < 60) return `${s.toFixed(1)}s`
   362→  const m = Math.floor(s / 60)
   363→  const rs = Math.floor(s % 60)
   364→  return `${m}m ${rs}s`
   365→}

   367→async function copyToClipboard(text: string): Promise<boolean> {
   368→  try {
   369→    if (navigator.clipboard && window.isSecureContext) {
   370→      await navigator.clipboard.writeText(text)
   371→    } else {
   372→      const ta = document.createElement('textarea')
   373→      ta.value = text
   374→      ta.style.position = 'fixed'
   375→      ta.style.opacity = '0'
   376→      document.body.appendChild(ta)
   377→      ta.select()
   378→      document.execCommand('copy')
   379→      document.body.removeChild(ta)
   380→    }
   381→    return true
   382→  } catch {
   383→    return false
   384→  }
   385→}

   853→function InfrastructureLogin({ onSuccess }: { onSuccess: (o: any) => void }) {

   990→// Inline SVG sparkline (normalised polyline) for metric cards.
   991→function Sparkline({ data, colour }: { data: number[]; colour: string }) {
   992→  const W = 120
   993→  const H = 36
   994→  if (!data || data.length < 2) {
   995→    return (
   996→      <svg width={W} height={H} className="opacity-40">
   997→        <line x1="0" y1={H - 1} x2={W} y2={H - 1} stroke="currentColor" strokeWidth="1" />
   998→      </svg>
   999→    )
  1000→  }
  1001→  const min = Math.min(...data)
  1002→  const max = Math.max(...data)
  1003→  const range = max - min || 1
  1004→  const stepX = W / (data.length - 1)
  1005→  const points = data
  1006→    .map((v, i) => {
  1007→      const x = i * stepX
  1008→      const y = H - 2 - ((v - min) / range) * (H - 4)
  1009→      return `${x.toFixed(1)},${y.toFixed(1)}`
  1010→    })
  1011→    .join(' ')
  1012→  return (
  1013→    <svg width={W} height={H} className="overflow-visible">
  1014→      <polyline
  1015→        points={points}
  1016→        fill="none"
  1017→        stroke={colour}
  1018→        strokeWidth="1.75"
  1019→        strokeLinecap="round"
  1020→        strokeLinejoin="round"
  1021→      />
  1022→      <circle
  1023→        cx={(data.length - 1) * stepX}
  1024→        cy={H - 2 - ((data[data.length - 1] - min) / range) * (H - 4)}
  1025→        r="2.5"
  1026→        fill={colour}
  1027→      />
  1028→    </svg>
  1029→  )
  1030→}
