   217→  const [demoBusy, setDemoBusy] = useState(false)
   218→  const [receiptCode, setReceiptCode] = useState('')
   219→  const [verifying, setVerifying] = useState(false)
   220→  const [verifyResult, setVerifyResult] = useState<any>(null)
   221→
   222→  useEffect(() => {
   223→    api.listOrganizations().then((d) => setOrgs(d.organizations || [])).catch(() => {})
   224→  }, [])
   225→
   226→  async function submitDemoRequest() {
   227→    if (!demoForm.name || !demoForm.email || !demoForm.org) {
   228→      toast.error('Please fill in your name, email, and organization.')
   229→      return
   230→    }
   231→    setDemoBusy(true)
   232→    try {
   233→      await api.submitTicket({
   234→        voterMatric: 'DEMO-REQUEST',
   235→        voterName: demoForm.name,
   236→        issueType: 'DEMO_REQUEST',
   237→        description: `Org: ${demoForm.org}\nContact: ${demoForm.name}\nEmail: ${demoForm.email}\nPhone: ${demoForm.phone || 'N/A'}\nOrg Type: ${demoForm.orgType || 'N/A'}\nEstimated Voters: ${demoForm.estimatedVoters || 'N/A'}\nPreferred Date: ${demoForm.preferredDate || 'N/A'}\n${demoForm.message || '(no message)'}`,
   238→      })
   239→      toast.success('Demo request received! Our team will contact you within 24 hours.')
   240→      setDemoForm({ name: '', email: '', org: '', phone: '', orgType: '', estimatedVoters: '', preferredDate: '', message: '' })
   241→    } catch {
   242→      toast.success('Demo request received! Our team will contact you within 24 hours.')
   243→      setDemoForm({ name: '', email: '', org: '', phone: '', orgType: '', estimatedVoters: '', preferredDate: '', message: '' })
   244→    } finally { setDemoBusy(false) }
   245→  }
   246→
   247→  async function verifyReceipt() {
   248→    if (!receiptCode.trim()) {
   249→      toast.error('Please enter your receipt code first.')
   250→      return
   251→    }
   252→    setVerifying(true); setVerifyResult(null)
   253→    try {
   254→      const d = await api.publicVerifyReceipt(receiptCode.trim())
   255→      setVerifyResult(d)
   256→      if (d.valid) toast.success('Receipt verified — your vote was counted!')
   257→    } catch (e: any) {
   258→      // The public endpoint returns 404 with a body for "not found" — the
   259→      // api helper throws on non-2xx, but the body is attached to err.data.
   260→      const payload = e?.data
   261→      if (payload && typeof payload === 'object' && 'valid' in payload) {
   262→        setVerifyResult(payload)
   263→      } else {
   264→        setVerifyResult({ valid: false, message: e?.message || 'Receipt not found.' })
   265→      }
   266→    } finally { setVerifying(false) }
   267→  }

   281→              {t('home.heroBadge')}
   282→            </Badge>
   283→            <h1 className="font-display text-4xl font-bold leading-[1.05] tracking-tight sm:text-5xl md:text-6xl">
   284→              {t('home.heroTitleLine1')}<br />

   286→              <span className="text-accent">{t('home.heroTitleLine3')}</span>

  1203→function HeroStat({ value, suffix, label }: { value: number; suffix?: string; label: string }) {
  1204→  const [display, setDisplay] = useState(0)
  1205→  useEffect(() => {
  1206→    const duration = 800
  1207→    const start = display
  1208→    const diff = value - start
  1209→    if (diff === 0) return
  1210→    const startTime = Date.now()
  1211→    const t = setInterval(() => {
  1212→      const elapsed = Date.now() - startTime
  1213→      const progress = Math.min(1, elapsed / duration)
  1214→      const eased = 1 - Math.pow(1 - progress, 3)
  1215→      setDisplay(Math.round(start + diff * eased))
  1216→      if (progress >= 1) clearInterval(t)
  1217→    }, 16)
  1218→    return () => clearInterval(t)
  1219→  }, [value])
  1220→  return (
  1221→    <div className="rounded-xl border border-border/60 bg-card/50 p-3 backdrop-blur">
  1222→      <div className="font-display text-2xl font-bold tabular-nums text-primary sm:text-3xl">
  1223→        {display.toLocaleString()}{suffix}
  1224→      </div>
  1225→      <div className="text-[10px] uppercase tracking-wider text-muted-foreground">{label}</div>
  1226→    </div>
  1227→  )
  1228→}
